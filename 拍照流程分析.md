**文件位置**: `camera/src/main/ets/cameramanagers/PhotoManager.ets:304-323`

```typescript
private calculateDeviceDegree(x: number, y: number, z: number): number {
  let deviceDegree: number = 0;
  // 判断设备是否接近垂直位置（垂直于地面）
  if ((x * x + y * y) * 3 < z * z) {
    return deviceDegree;
  } else {
    try {
      // 计算反正切值
      let sd: Decimal = Decimal.atan2(y, -x)
      // 将弧度值转换为角度值
      let sc: Decimal = Decimal.round(Number(sd) / Math.PI * 180);
      // 调整角度相对于垂直方向
      deviceDegree = 90 - Number(sc);
      // 将角度标准化到0-360度范围
      deviceDegree = deviceDegree >= 0 ? deviceDegree % 360 : deviceDegree % 360 + 360;
    } catch (exception) {
      Logger.error(TAG_LOG, `calculateDeviceDegree failed, code is ${exception.code}, message is ${exception.message}`);
    }
  }
  return deviceDegree;
}
```

async getOriginalImagePixelMap(photo: camera.Photo): Promise<image.PixelMap> {

                             watermarkInfo: WatermarkInfo, 
                             canvasWidth: number, 
                             canvasHeight: number) {
  
  // 1. 设置水印基础样式
  const watermarkConfig = {

    padding: 8,
    borderRadius: 8,
    position: { x: 16, y: 16 }  // 左上角定位
  };
  
  // 2. 绘制水印背景容器
  const watermarkWidth = 220;
  const watermarkHeight = 200;
  
  context.save();
  
  // 应用旋转和缩放变换（与WatermarkOverlay.ets保持一致）
  context.translate(watermarkConfig.position.x, watermarkConfig.position.y);

  await this.drawWatermarkContent(context, watermarkInfo, watermarkConfig);
  
  context.restore();

async compositeImageWithWatermark(originalPixelMap: image.PixelMap, 
                                 watermarkInfo: WatermarkInfo): Promise<image.PixelMap> {
  
  // 1. 获取原图尺寸
  const imageInfo = await originalPixelMap.getImageInfo();

  // 2. 创建离屏Canvas
  const offscreenCanvas = new OffscreenCanvas(width, height);

  // 4. 叠加水印
  await this.renderWatermarkOnCanvas(offscreenContext, watermarkInfo, width, height);
  
  // 5. 生成最终PixelMap
  const compositePixelMap = await offscreenContext.getPixelMap(0, 0, width, height);
  
  return compositePixelMap;
}

  imagePacker.release();
  accessHelper.release();
  compositePixelMap.release();

// 内存优化的图像处理器
class OptimizedImageProcessor {
  private static readonly MEMORY_LIMIT_MULTIPLIER: number = 2.0; // 官方基准
  private static readonly CHUNK_SIZE: number = 1024 * 1024; // 1MB块大小
  
  static async processWithMemoryConstraint(pixelMap: image.PixelMap, 
                                          watermarkInfo: WatermarkInfo): Promise<image.PixelMap> {
    const imageInfo: image.ImageInfo = await pixelMap.getImageInfo();
    const estimatedMemory: number = this.calculateMemoryUsage(imageInfo);
    const memoryLimit: number = imageInfo.size.width * imageInfo.size.height * 4 * this.MEMORY_LIMIT_MULTIPLIER;
    
    if (estimatedMemory > memoryLimit) {
      return this.processInChunks(pixelMap, watermarkInfo);
    }
    
    return this.processDirectly(pixelMap, watermarkInfo);
  }
  
  private static async processInChunks(pixelMap: image.PixelMap, 
                                      watermarkInfo: WatermarkInfo): Promise<image.PixelMap> {
    const imageInfo: image.ImageInfo = await pixelMap.getImageInfo();
    const chunkHeight: number = Math.floor(this.CHUNK_SIZE / (imageInfo.size.width * 4));
    
    // 创建输出PixelMap
    const outputPixelMap: image.PixelMap = await image.createPixelMapSync(
      new ArrayBuffer(imageInfo.size.width * imageInfo.size.height * 4), {
        size: imageInfo.size,
        pixelFormat: image.PixelMapFormat.RGBA_8888,
        editable: true
      }
    );
    
    // 分块处理
    for (let y = 0; y < imageInfo.size.height; y += chunkHeight) {
      const currentChunkHeight: number = Math.min(chunkHeight, imageInfo.size.height - y);
      
      // 处理当前块
      const chunkPixelMap: image.PixelMap = await this.extractChunk(pixelMap, y, currentChunkHeight);
      const processedChunk: image.PixelMap = await this.processChunkWithWatermark(chunkPixelMap, watermarkInfo, y);
      
      // 合并到输出
      await this.mergeChunk(outputPixelMap, processedChunk, y);
      
      // 立即释放块内存
      chunkPixelMap.release();
      processedChunk.release();
    }
    
    return outputPixelMap;
  }
  
  private static calculateMemoryUsage(imageInfo: image.ImageInfo): number {
    const baseSize: number = imageInfo.size.width * imageInfo.size.height * 4;
    return baseSize * 1.5; // 原图 + Canvas缓冲区
  }
}

class PerformanceMonitor {
  private static readonly metrics: PerformanceMetrics[] = [];
  
  static async measureWatermarkProcess<T>(
    operation: () => Promise<T>, 
    operationName: string
  ): Promise<T> {
    const startTime: number = performance.now();
    const startMemory: MemoryInfo = this.getMemoryInfo();
    
    const result: T = await operation();
    const endTime: number = performance.now();
    const endMemory: MemoryInfo = this.getMemoryInfo();
    
    const metrics: PerformanceMetrics = {
      operationName,
      executionTime: endTime - startTime,
      memoryUsage: endMemory.heapUsed - startMemory.heapUsed,
      timestamp: Date.now()
    };
    
    this.metrics.push(metrics);
    this.logPerformance(metrics);
    
    return result;
  }

      external: 0
    };

      `${metrics.operationName}: ${metrics.executionTime.toFixed(2)}ms, ` +
      `内存增长: ${(metrics.memoryUsage / 1024 / 1024).toFixed(2)}MB`
    );

mediaLibSavePhotoSingle(context: Context, imageObj: image.Image) {
  imageObj.getComponent(image.ComponentType.JPEG, async (errCode: BusinessError, component: image.Component) => {
    if (errCode || component === undefined) {
      Logger.error(TAG_LOG, 'getComponent failed');
      return;
    }
    const buffer: ArrayBuffer = component.byteBuffer;  // 获取JPEG字节缓冲区
    if (!buffer) {
      Logger.error(TAG_LOG, 'byteBuffer is null');
      return;
    }
    
    // 创建媒体资产请求
    let photoType: photoAccessHelper.PhotoType = photoAccessHelper.PhotoType.IMAGE;
    let extension: string = 'jpg';
    let options: photoAccessHelper.CreateOptions = {
      title: 'testPhoto'
    }
    
    try {
      let assetChangeRequest: photoAccessHelper.MediaAssetChangeRequest =
        photoAccessHelper.MediaAssetChangeRequest.createAssetRequest(context, photoType, extension, options);
      assetChangeRequest.addResource(photoAccessHelper.ResourceType.IMAGE_RESOURCE, buffer)
      assetChangeRequest.saveCameraPhoto();  // 标记为相机照片
      
      let accessHelper: photoAccessHelper.PhotoAccessHelper = photoAccessHelper.getPhotoAccessHelper(context);
      await accessHelper.applyChanges(assetChangeRequest);  // 应用更改到媒体库
      
      // 创建缩略图
      let imageSource = image.createImageSource(buffer);
      let pixelmap = imageSource.createPixelMapSync();
      this.callback(pixelmap, assetChangeRequest.getAsset().uri);  // 回调通知UI更新
      
      accessHelper.release();
      imageObj.release();
    } catch (exception) {
      Logger.error(TAG_LOG, `mediaLibSavePhotoSingle failed, code is ${exception.code}, message is ${exception.message}`);
    }
  });
}
